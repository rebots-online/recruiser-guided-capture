cmake_minimum_required(VERSION 3.22)
project(recruiser_depth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(RECRUISER_PARITY_TEST "Build standalone Vulkan/reference test, including on Android" OFF)
if(ANDROID)
    set(GLSLC "${CMAKE_ANDROID_NDK}/shader-tools/linux-x86_64/glslc")
else()
    find_program(GLSLC glslc REQUIRED)
endif()
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/unproject.inc"
    COMMAND "${GLSLC}" --target-env=vulkan1.0 -mfmt=c
        "${CMAKE_CURRENT_SOURCE_DIR}/unproject.comp" -o "${CMAKE_CURRENT_BINARY_DIR}/unproject.inc"
    DEPENDS unproject.comp VERBATIM)
if(ANDROID AND NOT RECRUISER_PARITY_TEST)
    add_library(recruiser_depth SHARED vulkan_unproject.cpp "${CMAKE_CURRENT_BINARY_DIR}/unproject.inc")
    target_link_libraries(recruiser_depth vulkan)
else()
    add_executable(recruiser_depth vulkan_unproject.cpp "${CMAKE_CURRENT_BINARY_DIR}/unproject.inc")
    target_compile_definitions(recruiser_depth PRIVATE RECRUISER_HOST_TEST=1)
    if(ANDROID)
        target_link_libraries(recruiser_depth vulkan)
    else()
        find_package(Vulkan REQUIRED)
        target_link_libraries(recruiser_depth Vulkan::Vulkan)
    endif()
    enable_testing()
    add_test(NAME vulkan_unprojection_parity COMMAND recruiser_depth)
endif()
target_include_directories(recruiser_depth PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(recruiser_depth PRIVATE -Wall -Wextra -Wno-missing-field-initializers -Werror=return-type)
